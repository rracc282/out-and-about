        <script data-turbo-eval-once="arena-megamenu">
(function() {
    // Scroll detection
    function handleScroll() {
        var navbar = document.getElementById('arenaNavbar');
        if (!navbar) return;
        if (window.scrollY > 50) {
            navbar.classList.add('is-scrolled');
        } else {
            navbar.classList.remove('is-scrolled');
        }
    }
    window.addEventListener('scroll', handleScroll, { passive: true });

    // Fermeture des dropdowns au clic ailleurs
    document.addEventListener('click', function() {
        var langSwitcher = document.getElementById('langSwitcher');
        var navItineraire = document.getElementById('navItineraire');
        if (langSwitcher) langSwitcher.classList.remove('is-open');
        if (navItineraire) navItineraire.classList.remove('is-open');
    });

    function initMegamenu() {
        if (document.body.dataset.arenaMegamenuInit === '1') return;
        document.body.dataset.arenaMegamenuInit = '1';

        handleScroll();

        var hamburger = document.getElementById('arenaHamburger');
        var mobileMenu = document.getElementById('arenaMobileMenu');
        var navItineraire = document.getElementById('navItineraire');
        var langSwitcher = document.getElementById('langSwitcher');

        // Itineraire + Language switcher
        if (navItineraire) {
            navItineraire.querySelector('.nav-itineraire__btn').addEventListener('click', function(e) {
                e.stopPropagation();
                navItineraire.classList.toggle('is-open');
                if (langSwitcher) langSwitcher.classList.remove('is-open');
            });
        }

        if (langSwitcher) {
            langSwitcher.querySelector('.lang-switcher__toggle').addEventListener('click', function(e) {
                e.stopPropagation();
                langSwitcher.classList.toggle('is-open');
                if (navItineraire) navItineraire.classList.remove('is-open');
            });
        }

        // Mobile menu toggle
        if (hamburger && mobileMenu) {
            hamburger.addEventListener('click', function() {
                hamburger.classList.toggle('is-open');
                mobileMenu.classList.toggle('is-open');
                document.body.classList.toggle('menu-open');
            });

            // Mobile submenu toggle
            mobileMenu.querySelectorAll('.arena-mobile-menu__toggle').forEach(function(btn) {
                btn.addEventListener('click', function() {
                    btn.classList.toggle('is-open');
                    var sublinks = btn.nextElementSibling;
                    if (sublinks) sublinks.classList.toggle('is-open');
                });
            });

            // Close on link click
            mobileMenu.querySelectorAll('a').forEach(function(link) {
                link.addEventListener('click', function() {
                    hamburger.classList.remove('is-open');
                    mobileMenu.classList.remove('is-open');
                    document.body.classList.remove('menu-open');
                });
            });
        }
    }

    document.addEventListener('page:render', initMegamenu);
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', initMegamenu);
    } else {
        initMegamenu();
    }
})();
